

## 8. 推荐压缩命令

以下命令只复制允许交付的内容，并在压缩前后检查规格卡：

```powershell
$source = 'D:\Desktop\codexTEST\_github_photo_check'
$zip = 'D:\Desktop\codexTEST\PhotoCheck_B集成测试包_2026-11-06.zip'
$staging = Join-Path $env:TEMP ('photocheck-b-release-' + [guid]::NewGuid())

$spec = Join-Path $source 'docs\功能规格卡_v1.4.md'
if (-not (Test-Path -LiteralPath $spec) -or (Get-Item -LiteralPath $spec).Length -eq 0) {
    throw '功能规格卡缺失或为空，终止压缩'
}

New-Item -ItemType Directory -Path $staging | Out-Null
robocopy $source $staging /E `
  /XD .git __pycache__ .pytest_cache .venv venv dist photocheck-output e2e-photocheck-output `
  /XF *.pyc *.log

if ($LASTEXITCODE -gt 7) {
    throw "robocopy failed: $LASTEXITCODE"
}

if (-not (Test-Path -LiteralPath (Join-Path $staging 'docs\功能规格卡_v1.4.md'))) {
    throw '暂存目录缺少功能规格卡，终止压缩'
}

Remove-Item -LiteralPath $zip -ErrorAction SilentlyContinue
Compress-Archive -Path (Join-Path $staging '*') -DestinationPath $zip

Add-Type -AssemblyName System.IO.Compression.FileSystem
$archive = [System.IO.Compression.ZipFile]::OpenRead($zip)
try {
    $hasSpec = $archive.Entries |
        Where-Object { $_.FullName -replace '\\', '/' -eq 'docs/功能规格卡_v1.4.md' }
    if (-not $hasSpec) {
        throw '压缩包中缺少 docs/功能规格卡_v1.4.md'
    }
}
finally {
    $archive.Dispose()
    Remove-Item -LiteralPath $staging -Recurse -Force
}

Write-Output $zip
```
